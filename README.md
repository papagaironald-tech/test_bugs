local function resolverCaminho(caminho)local atual=game local limpo=tostring(caminho or""):gsub("^%s+",""):gsub("%s+$","")limpo=limpo:gsub("^game[%./:]","")if limpo==""or limpo=="game"then return atual end for parte in limpo:gmatch("[^%./\\]+")do if atual==game then local ok,servico=pcall(function()return game:GetService(parte)end)if ok and servico then atual=servico else local achado=game:FindFirstChild(parte)if not achado then return nil,"não encontrado: "..parte end atual=achado end else local achado=atual:FindFirstChild(parte)if not achado then return nil,"não achei '"..parte.."' em "..atual:GetFullName()end atual=achado end end return atual end
